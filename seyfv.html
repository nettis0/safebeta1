async function load() { const TOKEN = atob(ENCRYPTED_TOKEN); const pass = document.getElementById('pass').value; // Добавили "?t=" + время, чтобы браузер не брал старую версию файла из кэша const res = await fetch(`https://api.github.com/repos/${REPO}/contents/${FILE}?t=${new Date().getTime()}`, { headers: { 'Authorization': `token ${TOKEN}`, 'Cache-Control': 'no-cache' } }); if (!res.ok) { alert("Ошибка загрузки: " + res.status); return; } const data = await res.json(); // Проверка: если файл пустой или нет контента if (!data.content) { alert("Файл на GitHub пустой или не найден"); return; } const raw = Uint8Array.from(atob(data.content), c => c.charCodeAt(0)); try { const decrypted = await crypto.subtle.decrypt( {name: "AES-GCM", iv: raw.slice(0, 12)}, await getCryptoKey(pass), raw.slice(12) ); document.getElementById('content').value = new TextDecoder().decode(decrypted); } catch(e) { alert("Ошибка расшифровки (неверный пароль?)"); console.error(e); } }
