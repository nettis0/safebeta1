async function save() { const pass = document.getElementById('pass').value; const text = document.getElementById('content').value; const key = await getCryptoKey(pass); const iv = crypto.getRandomValues(new Uint8Array(12)); const enc = await crypto.subtle.encrypt({name: "AES-GCM", iv}, key, new TextEncoder().encode(text)); // Преобразуем в base64 const content = btoa(String.fromCharCode(...iv, ...new Uint8Array(enc))); // Упрощенный запрос (PUT делает upsert) const response = await fetch(`https://api.github.com/repos/${REPO}/contents/${FILE}`, { method: 'PUT', headers: { 'Authorization': `token ${TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: "update content", content: content }) }); if (response.ok) { alert('Сохранено!'); } else { const err = await response.json(); alert('Ошибка: ' + JSON.stringify(err)); } } async function load() { const pass = document.getElementById('pass').value; const response = await fetch(`https://api.github.com/repos/${REPO}/contents/${FILE}`, { headers: {'Authorization': `token ${TOKEN}`, 'Cache-Control': 'no-cache'} }); if (!response.ok) { alert('Файл не найден или ошибка доступа'); return; } const res = await response.json(); const raw = Uint8Array.from(atob(res.content), c => c.charCodeAt(0)); try { const decrypted = await crypto.subtle.decrypt({name: "AES-GCM", iv: raw.slice(0, 12)}, await getCryptoKey(pass), raw.slice(12)); document.getElementById('content').value = new TextDecoder().decode(decrypted); } catch(e) { alert('Ошибка расшифровки (неверный пароль?)'); } }
